cmake_minimum_required(VERSION 3.20)
project(spacepharer VERSION 2.1.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# The pinned MMseqs headers rely on transitive stdint includes removed in GCC13+.
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-include> $<$<COMPILE_LANGUAGE:CXX>:cstdint>)
set(VERSION_OVERRIDE "2.1.1" CACHE STRING "Native engine version")
include(CTest)
set(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
set(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} "${CMAKE_CURRENT_SOURCE_DIR}/lib/mmseqs/cmake")

set(FRAMEWORK_ONLY 1 CACHE INTERNAL "" FORCE)
include(MMseqsSetupDerivedTarget)
add_subdirectory(lib/mmseqs)
add_subdirectory(data)
include_directories(lib)
add_subdirectory(src)

if(BUILD_TESTING)
  add_executable(spacepharer-statistics-tests tests/native/statistics.cpp)
  target_include_directories(spacepharer-statistics-tests PRIVATE src)
  target_compile_options(spacepharer-statistics-tests PRIVATE -UNDEBUG)
  add_test(NAME statistics COMMAND spacepharer-statistics-tests)
  add_executable(spacepharer-records-tests tests/native/records.cpp)
  target_include_directories(spacepharer-records-tests PRIVATE src)
  target_compile_options(spacepharer-records-tests PRIVATE -UNDEBUG)
  add_test(NAME records COMMAND spacepharer-records-tests)
endif()
